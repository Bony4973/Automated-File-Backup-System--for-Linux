" #!/bin/bash

# Backup Directory
BACKUP_DIR="/path/to/backup"   # Change to your backup destination
LOG_FILE="$BACKUP_DIR/backup.log"
TIMESTAMP=$(date +"%Y-%m-%d_%H-%M-%S")
BACKUP_FILE="backup_$TIMESTAMP.tar.gz"

# Directories/Files to Backup 
SOURCE_DIRS="/"  # Root directory to backup everything. Modify as needed.

# Create Backup Directory if it doesn't exist
if [ ! -d "$BACKUP_DIR" ]; then
    mkdir -p "$BACKUP_DIR"
fi

# Perform Backup
echo "[$(date)] Starting backup..." >> "$LOG_FILE"
tar -czvf "$BACKUP_DIR/$BACKUP_FILE" $SOURCE_DIRS >> "$LOG_FILE" 2>&1

# Backup Success/Failure
if [ $? -eq 0 ]; then
    echo "[$(date)] Backup successful: $BACKUP_FILE" >> "$LOG_FILE"
else
    echo "[$(date)] Backup failed" >> "$LOG_FILE"
fi

# Optional: Cleanup old backups (retain only last 7 backups)
echo "[$(date)] Cleaning up old backups..." >> "$LOG_FILE"
ls -1t "$BACKUP_DIR" | grep "backup_" | tail -n +8 | xargs -I {} rm -f "$BACKUP_DIR/{}"

echo "[$(date)] Backup complete." >> "$LOG_FILE" " explain this script.
